// Accessing Object properties

1. Dot
console.log(lightBulb.isOn);
console.log(lightBulb.toggle());

2. Bracket (same effect)
console.log(lightBulb['isOn']);
console.log(lightBulb['toggle']());

3. Keys with special chars -> bracket only
console.log(lightBulb['color-temp']);

4. Dynamic key
const key = 'isOn';
console.log(lightBulb[key]);

}
// to find sum of number dynamically
const sum = (...numbers) => {
    let total = 0;
    numbers.forEach(number => total += number)
    return total
}
