    console.log(count)
    count --
}

// for..of - values of an array/ iterable objects

// break & continue

for (let i=0; i<10; i++) {
    console.log(i);
    if(i ===5) {
        console.log('Stopping the loop at 5')
        break
    }
}

for (let i=0; i<10; i++) {
    if (i % 2 !== 0) {
        continue
    }
    console.log(i);
